function handleHoldClick(holdId) { if (!isPencilActive) return; if (!currentBlockName) { let newName = prompt("Création d'un nouveau bloc. Nom (ex: Mon Bloc (6A+)) :"); if (!newName) return; currentBlockName = newName; blocks[currentBlockName] = { depart: [], progression: [], arrivee: [], completed: false, noKickboard: chkEditKickboard.checked }; updateBlockSelect(); blockSelect.value = currentBlockName; // Mettre à jour l'apparence du bouton Save puisqu'on vient de créer un nom btnSave.innerText = "💾 Mettre à jour"; btnSave.classList.replace('btn-blue', 'btn-green'); } let block = blocks[currentBlockName]; // 1. On mémorise l'état actuel de la prise let estDepart = block.depart.includes(holdId); let estProgression = block.progression.includes(holdId); let estArrivee = block.arrivee.includes(holdId); // 2. On la nettoie de toutes les listes pour repartir à zéro block.depart = block.depart.filter(h => h !== holdId); block.progression = block.progression.filter(h => h !== holdId); block.arrivee = block.arrivee.filter(h => h !== holdId); // 3. Logique de cycle : Vide ➔ Vert ➔ Bleu ➔ Rouge ➔ Vide if (estDepart) { block.progression.push(holdId); // Si elle était verte, elle passe bleue } else if (estProgression) { block.arrivee.push(holdId); // Si elle était bleue, elle passe rouge } else if (estArrivee) { // Si elle était rouge, elle s'éteint (on ne la remet dans aucun tableau) } else { block.depart.push(holdId); // Si elle était vide, elle s'allume en vert } displayBlock(currentBlockName); }
